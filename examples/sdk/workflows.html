    Extract → LLM → email
    import OpenAI from "openai";

const data = doc.query.tables()
  .where({ headers: ["Deliverable", "Owner", "Due"] })
  .one().textRows;

const llm = new OpenAI();
const response = await llm.responses.create({
  model: process.env.OPENAI_MODEL,
  instructions: "Write a short team email summarizing " +
    "these deliveries. Keep every owner and date, " +
    "and ask the team to flag blockers.",
  input: JSON.stringify(data)
});

const email = response.output_text;
    Run this request on your server with OPENAI_API_KEY and OPENAI_MODEL configured. The live demo simulates this step; the code shows the actual LLM request.
